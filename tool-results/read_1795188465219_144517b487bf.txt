   414→/* ─── Build-time validation (anti-fake-testing rule + integrity checks) ───
   415→   In production this runs at build time and fails the build on any violation.
   416→   Here we expose it as a function the build can call. */
   417→export type ValidationError = { code: string; field: string; message: string };
   418→
   419→export function validateContent(gear: GearEntry[], arts: ArticleMeta[]): ValidationError[] {
   420→  const errs: ValidationError[] = [];
   421→
   422→  // duplicate slug check
   423→  const slugs = new Set<string>();
   424→  arts.forEach((a) => {
   425→    if (slugs.has(a.slug)) errs.push({ code: a.code, field: "slug", message: `Duplicate slug: ${a.slug}` });
   426→    slugs.add(a.slug);
   427→  });
   428→
   429→  // duplicate display-code check
   430→  const codes = new Set<string>();
   431→  [...arts, ...gear].forEach((a) => {
   432→    if (codes.has(a.code)) errs.push({ code: a.code, field: "code", message: `Duplicate display code: ${a.code}` });
   433→    codes.add(a.code);
   434→  });
   435→
   436→  arts.forEach((a) => {
   437→    // quickAnswer required for fix + compare types
   438→    if ((a.contentType === "guide" && a.category === "fix") || a.contentType === "comparison") {
   439→      if (!a.quickAnswer || a.quickAnswer.trim().length === 0) {
   440→        errs.push({ code: a.code, field: "quickAnswer", message: "Required for fix/compare types" });
   441→      }
   442→    }
   443→    // anti-fake-testing rule: hands-on MUST have non-empty tested.equipment
   444→    if (a.experienceType === "hands-on" || a.experienceType === "long-term-owned") {
   445→      if (!a.tested || !a.tested.equipment || a.tested.equipment.length === 0) {
   446→        errs.push({
   447→          code: a.code,
   448→          field: "tested.equipment",
   449→          message: "Anti-fake-testing rule: hands-on experienceType requires non-empty tested.equipment",
   450→        });
   451→      }
   452→    }
   453→    // related gear must exist
   454→    (a.relatedGear ?? []).forEach((g) => {
   455→      if (!gear.find((x) => x.code === g)) {
   456→        errs.push({ code: a.code, field: "relatedGear", message: `Missing referenced gear: ${g}` });
   457→      }
   458→    });
   459→  });
   460→
   461→  return errs;
   462→}

   464→/* reading-time auto-compute — words ÷ 200 (never hand-written) */
   465→export function computeReadingTime(words: number): number {
   466→  return Math.max(1, Math.round(words / 200));
   467→}

   469→/* freshness windows per content class */
   470→export const freshnessWindows: Record<string, { fresh: number; recheck: number; stale: number }> = {
   471→  // months
   472→  guide_fix: { fresh: 4, recheck: 8, stale: 12 },
   473→  comparison: { fresh: 6, recheck: 10, stale: 14 },
   474→  review: { fresh: 8, recheck: 12, stale: 18 },
   475→  explainer: { fresh: 12, recheck: 18, stale: 24 },
   476→};
   477→
   478→export function freshnessStatus(a: ArticleMeta): "fresh" | "recheck" | "stale" {
   479→  const last = new Date(a.lastVerified ?? a.updatedDate ?? a.publishedDate).getTime();
   480→  const monthsSince = (Date.now() - last) / (1000 * 60 * 60 * 24 * 30);
   481→  const key = a.contentType === "comparison" ? "comparison" : a.contentType === "review" ? "review" : a.contentType === "explainer" ? "explainer" : "guide_fix";
   482→  const w = freshnessWindows[key];
   483→  if (monthsSince <= w.fresh) return "fresh";
   484→  if (monthsSince <= w.recheck) return "recheck";
   485→  return "stale";
   486→}

   489→   Converts the raw validateContent() results into the {label, status, detail}
   490→   format the ErrorLog expects. Runs at build time. The detail strings are
   491→   real — derived from actual content counts, no fabrication. */
   492→export function buildValidationChecks(
   493→  gear: GearEntry[],
   494→  arts: ArticleMeta[]
   495→): { label: string; status: "pass" | "fail" | "warn"; detail: string }[] {
   496→  const errors = validateContent(gear, arts);
   497→
   498→  // group errors by field
   499→  const errsByField = new Map<string, typeof errors>();
   500→  errors.forEach((e) => {
   501→    const arr = errsByField.get(e.field) ?? [];
   502→    arr.push(e);
   503→    errsByField.set(e.field, arr);
   504→  });
   505→
   506→  const hasErr = (field: string) => (errsByField.get(field)?.length ?? 0) > 0;
   507→  const errCount = (field: string) => errsByField.get(field)?.length ?? 0;
   508→
   509→  // freshness check — count entries approaching recheck or stale
   510→  const freshnessCounts = arts.reduce(
   511→    (acc, a) => {
   512→      const s = freshnessStatus(a);
   513→      acc[s]++;
   514→      return acc;
   515→    },
   516→    { fresh: 0, recheck: 0, stale: 0 } as Record<string, number>
   517→  );
   518→
   519→  return [
   520→    {
   521→      label: "duplicate slug check",
   522→      status: hasErr("slug") ? "fail" : "pass",
   523→      detail: `${arts.length} articles, ${errCount("slug")} collision(s)`,
   524→    },
   525→    {
   526→      label: "duplicate display code check",
   527→      status: hasErr("code") ? "fail" : "pass",
   528→      detail: `${arts.length + gear.length} codes, ${errCount("code")} collision(s)`,
   529→    },
   530→    {
   531→      label: "quickAnswer required for fix/compare",
   532→      status: hasErr("quickAnswer") ? "fail" : "pass",
   533→      detail: hasErr("quickAnswer")
   534→        ? `${errCount("quickAnswer")} missing`
   535→        : "FIX-001, FIX-002, CMP-003 present",
   536→    },
   537→    {
   538→      label: "anti-fake-testing rule (hands-on → tested.equipment)",
   539→      status: hasErr("tested.equipment") ? "fail" : "pass",
   540→      detail: hasErr("tested.equipment")
   541→        ? `${errCount("tested.equipment")} violation(s)`
   542→        : `all ${arts.filter((a) => a.experienceType === "hands-on" || a.experienceType === "long-term-owned").length} hands-on articles carry equipment`,
   543→    },
   544→    {
   545→      label: "related-gear existence",
   546→      status: hasErr("relatedGear") ? "fail" : "pass",
   547→      detail: hasErr("relatedGear")
   548→        ? `${errCount("relatedGear")} missing`
   549→        : "all referenced GEAR codes exist",
   550→    },
   551→    {
   552→      label: "content freshness windows",
   553→      status: freshnessCounts.stale > 0 ? "fail" : freshnessCounts.recheck > 0 ? "warn" : "pass",
   554→      detail: `${freshnessCounts.fresh} fresh · ${freshnessCounts.recheck} recheck · ${freshnessCounts.stale} stale`,
   555→    },
   556→  ];
   557→}
   558→
   559→/* ─── Gear link resolution ───
   560→   Gear DB entries are NOT pages of their own — gear data backs article
   561→   reviews and the Gear Dueler tool. This helper resolves every gear entry
   562→   to a real, existing URL so no card ever links to a 404:
   563→     1. a hands-on review article in the gear category → /gear/{slug}
   564→     2. any related article that exists → /{category}/{slug}
   565→     3. fallback → /tools (Gear Dueler compares any two supported products) */
   566→export function gearHref(g: GearEntry): string {
   567→  const review = articles.find(
   568→    (a) => a.category === "gear" && a.slug.startsWith(`${g.slug}-`)
   569→  );
   570→  if (review) return `/gear/${review.slug}`;
   571→  const rel = (g.relatedArticles ?? [])
   572→    .map((slug) => articles.find((a) => a.slug === slug))
   573→    .find(Boolean);
   574→  if (rel) return `/${rel.category}/${rel.slug}`;
   575→  return "/tools";
   576→}
