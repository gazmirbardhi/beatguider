    15→export function PanelButton({
    16→  children,
    17→  href,
    18→  variant = "default",
    19→  onClick,
    20→  type = "button",
    21→  className = "",
    22→  ariaLabel,
    23→  target,
    24→  rel,
    25→}: {
    26→  children: ReactNode;
    27→  href?: string;

    31→  className?: string;
    32→  ariaLabel?: string;
    33→  target?: string;
    34→  rel?: string;
    35→}) {
    36→  const cls = `hwbtn ${variant === "amber" ? "hwbtn-amber" : ""} ${className}`;

    43→  }
    44→  return (
    45→    <button type={type} onClick={onClick} className={cls} aria-label={ariaLabel}>
    46→      {children}
    47→    </button>
    48→  );
    49→}

   262→/* ─── Checklist: physical toggle switch, green when on, strikes text ─── */
   263→export function Checklist({ items: initial }: { items: { id: string; text: string }[] }) {
   264→  const [state, setState] = useState<Record<string, boolean>>({});
   265→  const toggle = (id: string) => setState((s) => ({ ...s, [id]: !s[id] }));
   266→
   267→  return (
   268→    <ul className="my-5 space-y-1.5">
   269→      {initial.map((it) => {
   270→        const on = !!state[it.id];
   271→        return (
   272→          <li
   273→            key={it.id}
   274→            className="flex items-center gap-3 p-2.5 border rounded-sm"
   275→            style={{ borderColor: "var(--line)", background: "var(--plate)" }}
   276→          >
   277→            <button
   278→              type="button"
   279→              className="togglesw"
   280→              data-on={on}
   281→              aria-pressed={on}
   282→              aria-label={`${on ? "Mark incomplete" : "Mark complete"}: ${it.text}`}
   283→              onClick={() => toggle(it.id)}
   284→            />
   285→            <span
   286→              className={`text-sm flex-1 ${on ? "line-through" : ""}`}
   287→              style={{ color: on ? "var(--mut)" : "var(--body)" }}
   288→            >
   289→              {it.text}
   290→            </span>
   291→            {on && <span className="led led-g" aria-hidden="true" />}
   292→          </li>
   293→        );
   294→      })}
   295→    </ul>
   296→  );
   297→}

   419→/* ─── VU meter: 26 segments, green→amber→red at 16/21 ─── */
   420→export function VUMeter({
   421→  value,
   422→  segments = 26,
   423→  showDigits = true,
   424→  size = "md",
   425→}: {
   426→  value: number; // 0..100
   427→  segments?: number;
   428→  showDigits?: boolean;
   429→  size?: "sm" | "md" | "lg";
   430→}) {
   431→  const v = Math.max(0, Math.min(100, value));
   432→  const lit = Math.round((v / 100) * segments);
   433→  const h = size === "sm" ? "h-2" : size === "lg" ? "h-4" : "h-3";
   434→  return (
   435→    <div className="electronics electronics-deep p-3 sm:p-4 my-4">
   436→      <div className="flex items-end gap-3">
   437→        <div className="flex items-center gap-px flex-1 vu-needle-host" aria-hidden="true">
   438→          {Array.from({ length: segments }).map((_, i) => {
   439→            const on = i < lit;
   440→            const cls = i < 16 ? "vu-seg-g" : i < 21 ? "vu-seg-a" : "vu-seg-r";
   441→            return <span key={i} className={`vu-seg ${h} ${on ? cls : ""}`} />;
   442→          })}
   443→          {/* analog needle overlay — points to the current level */}
   444→          <span
   445→            className="vu-needle"
   446→            style={{ left: `${(lit / segments) * 100}%` }}
   447→          />
   448→        </div>
   449→        {showDigits && (
   450→          <div className="flex flex-col items-end flex-shrink-0">
   451→            <span className="silk !text-[0.5625rem]" style={{ color: "rgba(255,176,0,0.55)" }}>LEVEL</span>
   452→            <span className="lcd lcd-ghost !text-lg !bg-transparent font-semibold tabular-nums">
   453→              {String(Math.round(v)).padStart(3, "0")}%
   454→            </span>
   455→          </div>
   456→        )}
   457→      </div>
   458→    </div>
   459→  );
   460→}

   965→/* ─── Diagnostic sequence — signal-path stepper for FIX articles ───
   966→   A horizontal flow diagram showing the diagnostic steps as a signal
   967→   chain, like a real test-bench signal path. Each node is a dark display
   968→   module with the step number glowing amber. Active step (click) lights
   969→   amber + scrolls to the matching section. Reduced-motion safe. */
   970→export function DiagnosticSequence({
   971→  steps,
   972→  active: activeProp,
   973→  onSelect,
   974→}: {
   975→  steps: { n: number; label: string; status?: "pass" | "fail" | "pending"; target?: string }[];
   976→  active?: number;
   977→  onSelect?: (n: number) => void;
   978→}) {
   979→  const [activeState, setActiveState] = useState<number | undefined>(activeProp);
   980→  const active = activeProp ?? activeState;
   981→
   982→  const handleSelect = (s: { n: number; target?: string }) => {
   983→    setActiveState(s.n);
   984→    onSelect?.(s.n);
   985→    // scroll to the target section if provided (TOC chip behavior)
   986→    if (s.target) {
   987→      const el = document.getElementById(s.target);
   988→      if (el) {
   989→        el.scrollIntoView({ behavior: "smooth", block: "start" });
   990→      }
   991→    }
   992→  };
   993→
   994→  return (
   995→    <nav aria-label="Diagnostic sequence" className="my-6 electronics p-4">
   996→      <div className="flex items-center gap-2 mb-3">
   997→        <span className="led led-a led-breathe" aria-hidden="true" />
   998→        <span className="silk" style={{ color: "var(--amber)" }}>Signal path · diagnostic sequence</span>
   999→        <span className="ml-auto silk !text-[0.5625rem]" style={{ color: "rgba(255,176,0,0.55)" }}>
  1000→          {active ? `STEP ${String(active).padStart(2, "0")} ACTIVE` : "click to jump"}
  1001→        </span>
  1002→      </div>
  1003→      <ol className="flex items-stretch gap-1 overflow-x-auto scroll-x pb-1">
  1004→        {steps.map((s, i) => {
  1005→          const isActive = active === s.n;
  1006→          const statusLed =
  1007→            s.status === "pass" ? "led led-g" :
  1008→            s.status === "fail" ? "led led-r" :
  1009→            isActive ? "led led-a" : "led";
  1010→          return (
  1011→            <li key={s.n} className="flex items-stretch flex-shrink-0">
  1012→              <button

  1173→/* ─── Share button — copy-to-clipboard with hardware-button styling ───
